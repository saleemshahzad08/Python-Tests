// basics & setup

Minimal Window

from tkinter import Tk root = Tk() root.title("My App") root.geometry("400x300") root.mainloop()

Window Configuration

Method Purpose Example
root.title() Set window title root.title("App Name")
root.geometry() Set size & position root.geometry("400x300+100+50")
root.config() Configure properties root.config(bg="white")
root.resizable() Allow/disable resize root.resizable(False, False)
root.mainloop() Keep window open root.mainloop()

Import Patterns

from tkinter import Tk, Label, Button, Entry from tkinter.ttk import Combobox, Spinbox from tkinter import messagebox, filedialog
// all widgets quick reference

Label - Display Text

label = Label(root, text="Hello", font=("Arial", 12), fg="black", bg="white") label.pack()

Button - Clickable Element

def on_click(): print("Clicked!") button = Button(root, text="Click", command=on_click) button.pack()

Entry - Single Line Input

entry = Entry(root, width=30) entry.pack() value = entry.get() entry.delete(0, "end") entry.insert(0, "default")

Text - Multi-line Input

text = Text(root, height=10, width=40) text.pack() content = text.get("1.0", "end-1c") text.insert("1.0", "Hello") text.delete("1.0", "end")

Frame - Container

frame = Frame(root, bg="lightblue", height=100, width=300) frame.pack() label = Label(frame, text="Inside frame") label.pack()

Listbox - List Selection

listbox = Listbox(root, height=8, width=30) listbox.pack() listbox.insert(0, "Item 1") listbox.insert(1, "Item 2") selected = listbox.curselection()

Checkbutton - Multiple Selection

from tkinter import IntVar var = IntVar() check = Checkbutton(root, text="Option", variable=var) check.pack() is_checked = var.get() # 1 or 0

Radiobutton - Single Selection

from tkinter import StringVar var = StringVar(value="default") radio1 = Radiobutton(root, text="A", variable=var, value="A") radio1.pack() selected = var.get()

Combobox - Dropdown

from tkinter.ttk import Combobox combo = Combobox(root, values=["A", "B", "C"]) combo.pack() combo.set("A") selected = combo.get()

Scale - Slider

from tkinter import Scale, HORIZONTAL scale = Scale(root, from_=0, to=100, orient=HORIZONTAL) scale.pack() value = scale.get()

Spinbox - Numeric Input

from tkinter.ttk import Spinbox spinbox = Spinbox(root, from_=0, to=100) spinbox.pack() value = spinbox.get()

Menu - Menu Bar

menubar = Menu(root) root.config(menu=menubar) file_menu = Menu(menubar, tearoff=0) menubar.add_cascade(label="File", menu=file_menu) file_menu.add_command(label="Open", command=func) file_menu.add_separator() file_menu.add_command(label="Exit", command=root.quit)

Common Properties (All Widgets)

Property Values/Type Example
text string text="Hello"
font (name, size, style) font=("Arial", 12, "bold")
fg color name or hex fg="red" or fg="#FF0000"
bg color name or hex bg="lightblue"
padx integer (pixels) padx=10
pady integer (pixels) pady=5
width integer width=20
height integer height=10
state "normal" or "disabled" state="disabled"
command function command=my_func

Font Styles

font=("Arial", 14) - Normal
font=("Arial", 14, "bold") - Bold
font=("Arial", 14, "italic") - Italic
font=("Arial", 14, "bold italic") - Bold Italic

Common Colors

Named: "white", "black", "red", "blue", "green", "yellow", "lightblue", "lightgreen", "gray", "orange", "purple"
Hex: "#FF0000" (red), "#00FF00" (green), "#0000FF" (blue)
// widget methods

Entry Methods

Method Purpose Example
.get() Get text value = entry.get()
.insert() Insert text entry.insert(0, "Hello")
.delete() Delete text entry.delete(0, "end")

Text Methods

Method Purpose Example
.get() Get range of text text.get("1.0", "end")
.insert() Insert text at position text.insert("1.0", "Hello")
.delete() Delete text range text.delete("1.0", "end")

Listbox Methods

Method Purpose Example
.insert() Add item listbox.insert("end", "Item")
.delete() Remove item listbox.delete(0)
.get() Get item text listbox.get(0)
.curselection() Get selected index selected = listbox.curselection()

Var Methods (StringVar, IntVar)

Method Purpose Example
.get() Get value value = var.get()
.set() Set value var.set("new value")
// layout managers
📦 Pack Layout
▼

Stacks widgets in a line (top-to-bottom or left-to-right)

widget.pack(side="top", fill="x", expand=True, padx=10, pady=5)
Parameter Values
side "top", "bottom", "left", "right"
fill "x", "y", "both", "none"
expand True, False
padx integer
pady integer
🎯 Grid Layout (PROFESSIONAL)
▼

Arranges widgets in rows and columns (like a table)

widget.grid(row=0, column=0, columnspan=2, rowspan=1, sticky="nsew", padx=5, pady=5)
Parameter Purpose
row Row number (0-based)
column Column number (0-based)
columnspan Span N columns
rowspan Span N rows
sticky "n", "s", "e", "w", "ne", "nsew"
padx Horizontal padding
pady Vertical padding
📍 Place Layout
▼

Positions widgets at exact pixel coordinates

widget.place(x=100, y=50, width=150, height=50)
Parameter Purpose
x X coordinate (pixels)
y Y coordinate (pixels)
width Width (pixels)
height Height (pixels)
relx Relative X (0.0-1.0)
rely Relative Y (0.0-1.0)
// event handling & shortcuts

Event Binding

def on_event(event): print(f"Event: {event}") widget.bind("", on_event) # Left click widget.bind("", on_event) # Right click widget.bind("", on_event) # Double click widget.bind("", on_event) # Mouse enter widget.bind("", on_event) # Mouse leave widget.bind("", on_event) # Mouse move

Keyboard Events

root.bind("", on_enter) # Enter key root.bind("", on_space) # Space bar root.bind("", on_key) # Any key root.bind("", on_save) # Ctrl+S root.bind("", on_open) # Ctrl+O

Lambda with Arguments

def greet(name): print(f"Hello, {name}!") button = Button(root, text="Greet", command=lambda: greet("Alice")) button.pack()

// common patterns

Class-Based Application

from tkinter import Tk, Label, Button class MyApp: def __init__(self, root): self.root = root self.root.title("My App") self.create_widgets() def create_widgets(self): self.label = Label(self.root, text="Hello") self.label.pack() self.button = Button(self.root, text="Click", command=self.on_click) self.button.pack() def on_click(self): self.label.config(text="Clicked!") if __name__ == "__main__": root = Tk() app = MyApp(root) root.mainloop()

File Operations

try: with open("file.txt", "r") as file: content = file.read() except FileNotFoundError: messagebox.showerror("Error", "File not found") except Exception as e: messagebox.showerror("Error", f"Error: {str(e)}") try: with open("file.txt", "w") as file: file.write(content) except Exception as e: messagebox.showerror("Error", f"Cannot save: {e}")

Getting Multiple Entry Values

name_entry = Entry(root) email_entry = Entry(root) age_entry = Entry(root) def submit(): name = name_entry.get() email = email_entry.get() age = age_entry.get() if name and email and age: print(f"Name: {name}, Email: {email}, Age: {age}") else: messagebox.showwarning("Error", "Fill all fields!") button = Button(root, text="Submit", command=submit)

Calculator Pattern

display = Entry(root, font=("Arial", 20)) display.pack() def append_number(num): current = display.get() if current == "0": display.delete(0, "end") display.insert("end", str(num)) def calculate(): try: result = eval(display.get()) display.delete(0, "end") display.insert(0, str(result)) except: messagebox.showerror("Error", "Invalid expression") button1 = Button(root, text="1", command=lambda: append_number(1)) button1.pack()
// geometry format

Window Geometry String

Format: "WIDTHxHEIGHT+X+Y"

Examples:
"400x300" - 400px wide, 300px tall
"600x400+100+50" - 600×400, positioned at (100,50)
"500x500" - 500×500, auto-positioned

Text Widget Positions

"1.0" - Line 1, Character 0 (start)
"end" - End of text
"end-1c" - End minus 1 character (removes newline)
"2.5" - Line 2, Character 5
