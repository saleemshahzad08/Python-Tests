// combobox widget

What is a Combobox?

A Combobox is a dropdown list that lets users select from predefined options OR type their own value.

Creating a Combobox

from tkinter import Tk from tkinter.ttk import Combobox root = Tk() root.title("Combobox Demo") root.geometry("400x300") combo = Combobox(root, values=["Python", "Java", "C++", "JavaScript"]) combo.pack(padx=10, pady=10) combo.set("Select a language") root.mainloop()

Getting Combobox Value

from tkinter import Tk, Button from tkinter.ttk import Combobox root = Tk() root.title("Combobox Demo") root.geometry("400x300") combo = Combobox(root, values=["Python", "Java", "C++"]) combo.pack(padx=10, pady=10) combo.set("Select") def get_selection():   selected = combo.get()   print(f"You selected: {selected}") button = Button(root, text="Get Selection", command=get_selection) button.pack() root.mainloop()

Combobox Properties

Property Purpose Example
values List of options values=["A", "B", "C"]
state "readonly" or "normal" state="readonly"
width Width in characters width=20
// menu & menubar

Creating a Menu Bar

from tkinter import Tk, Menu root = Tk() root.title("Menu Demo") root.geometry("400x300") menubar = Menu(root) root.config(menu=menubar) # File Menu file_menu = Menu(menubar, tearoff=0) menubar.add_cascade(label="File", menu=file_menu) file_menu.add_command(label="New", command=lambda: print("New")) file_menu.add_command(label="Open", command=lambda: print("Open")) file_menu.add_separator() file_menu.add_command(label="Exit", command=root.quit) # Edit Menu edit_menu = Menu(menubar, tearoff=0) menubar.add_cascade(label="Edit", menu=edit_menu) edit_menu.add_command(label="Cut", command=lambda: print("Cut")) edit_menu.add_command(label="Copy", command=lambda: print("Copy")) edit_menu.add_command(label="Paste", command=lambda: print("Paste")) root.mainloop()

Menu Methods

Method Purpose Example
.add_command() Add clickable item menu.add_command(label="Open", command=func)
.add_separator() Add divider line menu.add_separator()
.add_cascade() Add submenu menubar.add_cascade(label="File", menu=file_menu)
// scale & spinbox widgets

Scale Widget (Slider)

from tkinter import Tk, Scale, HORIZONTAL root = Tk() root.title("Scale Demo") root.geometry("400x300") # Horizontal scale scale = Scale(root, from_=0, to=100, orient=HORIZONTAL) scale.pack(padx=10, pady=10) def get_value():   value = scale.get()   print(f"Scale value: {value}") from tkinter import Button button = Button(root, text="Get Value", command=get_value) button.pack() root.mainloop()

Spinbox Widget

from tkinter import Tk, Button from tkinter.ttk import Spinbox root = Tk() root.title("Spinbox Demo") root.geometry("400x300") # Numeric spinbox spinbox = Spinbox(root, from_=0, to=100) spinbox.pack(padx=10, pady=10) def get_value():   value = spinbox.get()   print(f"Spinbox value: {value}") button = Button(root, text="Get Value", command=get_value) button.pack() root.mainloop()
// message boxes & dialogs

Message Box Functions

from tkinter import Tk, Button, messagebox root = Tk() root.title("Message Boxes") root.geometry("400x300") def show_info():   messagebox.showinfo("Info", "This is info") def show_warning():   messagebox.showwarning("Warning", "This is a warning") def show_error():   messagebox.showerror("Error", "This is an error") def ask_yes_no():   response = messagebox.askyesno("Question", "Do you like Python?")   print(f"Response: {response}") info_btn = Button(root, text="Info", command=show_info) info_btn.pack(pady=5) warning_btn = Button(root, text="Warning", command=show_warning) warning_btn.pack(pady=5) error_btn = Button(root, text="Error", command=show_error) error_btn.pack(pady=5) yes_no_btn = Button(root, text="Yes/No", command=ask_yes_no) yes_no_btn.pack(pady=5) root.mainloop()

File Dialogs

from tkinter import Tk, Button, filedialog root = Tk() root.title("File Dialogs") root.geometry("400x300") def open_file():   file_path = filedialog.askopenfilename(     title="Open File",     filetypes=[("Text Files", "*.txt"), ("All Files", "*.*")]   )   if file_path:     print(f"Selected: {file_path}") def save_file():   file_path = filedialog.asksaveasfilename(     title="Save File",     defaultextension=".txt",     filetypes=[("Text Files", "*.txt"), ("All Files", "*.*")]   )   if file_path:     print(f"Saving to: {file_path}") open_btn = Button(root, text="Open File", command=open_file) open_btn.pack(pady=10) save_btn = Button(root, text="Save File", command=save_file) save_btn.pack(pady=10) root.mainloop()

Message Box Functions

Function Purpose Returns
showinfo() Information dialog None
showwarning() Warning dialog None
showerror() Error dialog None
askyesno() Yes/No question True/False
askokcancel() OK/Cancel question True/False
// key takeaways
  • Combobox provides dropdown selections with editing
  • Menu/Menubar creates professional menus
  • Scale is a slider for numeric input
  • Spinbox is numeric input with arrow buttons
  • Message boxes provide dialog interactions
  • File dialogs let users open/save files
  • Combobox is from tkinter.ttk, not tkinter
